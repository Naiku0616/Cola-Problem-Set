貌似懂了，栈是一个概念，核心是先进后出，可以通过不同方式来实现。这道题可以用string来模拟栈，也只是模拟，不是真正的栈，string没有栈的强制先进后出规则（又研究了好一会儿什么是栈）

### 关键点1
怎么把字符串里的"niu"或"mo"找出来并删除
#### 思路1
把字符串依次压进栈内，每次压进一个字符就检查栈顶的前2个或3个元素来实现"niu"和"mo"的查找，找到后把栈里相应的那几个元素全部弹出，以此类推直到把全部字符都在栈里过一遍，看看最后栈是否是空的就能判断能不能删光整个字符串

```C++
#include <bits/stdc++.h>
using namespace std;

int main() {

    int n;
    string s;
    cin >> n >> s;

    string stk;
    for (char ch : s) { 
        //(char ch : s)相当于 for (int i = 0; i < s.size(); i++) { char ch = s[i];} 
        //还挺方便

        //将字符ch压入栈中
        stk.push_back(ch);
        int len = stk.size();

        //检查栈顶的字符是否满足条件，如果满足则弹出相应的字符
        //查找"niu"
        if (len >= 3 && stk[len-3] == 'n' && stk[len-2] == 'i' && stk[len-1] == 'u') {
            stk.pop_back();
            stk.pop_back();
            stk.pop_back();
        }
        //查找"mo"
        else if (len >= 2 && stk[len-2] == 'm' && stk[len-1] == 'o') {
            stk.pop_back();
            stk.pop_back();
        }

    }
    //检查栈是否为空，空则输出"Yes"，否则输出"No"
    if(stk.empty()) {
        cout << "Yes";
    }
    else {
        cout << "No";
    }

    return 0;
}
```
