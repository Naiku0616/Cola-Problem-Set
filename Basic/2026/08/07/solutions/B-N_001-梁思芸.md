#include <iostream>
#include<algorithm>
using namespace std;
int main()
{
int t,n,a[60];
cin>>t;
int i;
for(i=0;i<t;i++)
{
	cin>>n;
	int sum=0;
	int val=0;
	int m=0;
	for(int k=0;k<n;k++)
	{
		cin>>a[k];
		sum=sum+a[k];
	}
	for(int k=0;k<n;k++)
	{
		int amount=0;
			for(int j=0;j<n;j++) 
		    {
			if(a[j]==a[k])
			{
			amount=amount+1;
		    }
		    }
		if(amount>m)
		{
			m=amount;
			val=a[k];
		}
	}
	int x;
	int point;
	 x=n-m;
	if(m-x>2)
	{
		point=sum-val*m+val*(x+2);
	}
	else
	{
		point=sum;
		
	}
cout<<point<<endl;
}
return 0;
}
