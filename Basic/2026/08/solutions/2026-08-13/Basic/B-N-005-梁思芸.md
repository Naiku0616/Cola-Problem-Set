#include <iostream>
#include <cstdlib>
#include <vector>
using namespace std;
int main()
{
	int t,n;
	cin>>t;
	for(int i = 0;i < t;i++)
	{
		cin>>n;
		if (n == 1)
		{
		cout<<"NO"<<endl;
		continue;
	    }
		vector<long long>w(n);
		for(int j = 0;j < n;j++)
		{
			cin>>w[j];
			bool ok = true;
			if (abs(w[j+1]-w[j])<2)
			{
				ok=false;
				break;
			}
		cout<<(ok?"YES":"NO")<<endl;
		}
		if(n == 1)
		{
		cout<<"NO"<<endl;
	    }
	}
	return 0;
}
