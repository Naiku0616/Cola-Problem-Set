/* 思路一：
* 对于数组而言（数组内含有3个及以上变量），任意数组的排列方式只有有序与无序
* 对于无序的数组而言，相邻项的差值的和必定会大于有序的数组
* 所以，存在任意的有序数组之间的差值的和必定小于无序数组之间的差值的和
* 在有序的数组中存在升序数组与降序数组，其之间的差值的和满足相反数的特征
* 存在绝对值的前提下，其相反数的值，输出也与升序数组的值相同
* 即存在两个相同的最小值
*/

// 当然还存在一种各项数值都相同的变量，这种在题目的要求下，无论如何都只有一个最小值，且肯定为0

```cpp
#include <bits/stdc++.h>
#include <algorithm>            //引入排序组合，默认是升序
using namespace std;
int main(){
    int num ;
    cin >> num;
    int v[num];
    for (int i=0 ;i<num; i++){
        cin >> v[i];
    }
    // 数组内容的输入
    int n =sizeof(v)/sizeof(v[0]);
    //对数组本身进行重排序
    //需要遍历数组中每一个元素，通过取值进行加减比较，最终得到最小值，且判断新数组中存在多少个最小值
    sort (v,v+n);
    // 重新对v数组进行按升序排列
    int num1=0;
        for (int i1=n-1;i1>0;i1--){
        // 对数组各项进行相邻项的减法
            num1=v[i1]-v[i1-1]+num1;
        }
            //遍历的情况不允许出现自己减自己
        // 根据实验，对于数组中各个数据均相等的前提下，该数组有且仅有一种排列方式
        // 对于数组中存在一个数与其他数均不相同，则该数组有且仅有两种排列方式
            if (num1=0){
                cout << "1 " << num1 <<endl;
            }
            else {
            cout << "2 " << num1 << endl;
            }
                //num1代表最小值大小
    return 0;
}
```
