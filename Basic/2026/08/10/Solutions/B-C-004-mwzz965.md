经过和ds斗智斗勇下面是题解  
  
1.找到数量最多的数num，并记录它的数量cnt  
2.拿剩下的数rest当隔板，num最多被rest + 1个数隔开  
3.隔板够多即rest + 1 >= cnt时，全都可以用，答案就是和  
4.隔板不够即rest + 1 < cnt时，必然有数被浪费，且被浪费的必然是数量最多的数，数量为(cnt - (rest + 1) - 1)，因此可以sum - waste

```cpp
#include <bits/stdc++.h>
using namespace std;

int t;

int main(){

    cin >> t;
    for (int i = 0; i < t; i++)
    {
        int n, sum = 0, ans;
        cin >> n;
        map<int, int> mp;

        for (int i = 0; i < n; i++){
            int temp;
            cin >> temp;
            mp[temp]++;
            sum += temp;
        }

        int cnt = 0, num, rest;

        for(auto [x, y] : mp){
            if(cnt < y){
                cnt = y;
                num = x;
            }
        }
        //找到num和cnt

        rest = n - cnt;

        if(cnt <= rest + 1) ans = sum;
        else{
            ans = sum - (cnt - rest - 2) * num;
        }
        cout << ans << endl;
    }

    return 0;
}
```
